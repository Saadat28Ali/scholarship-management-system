Return structure (all endpoints): {
	"success": True,
	"msg": "",
	"details": {
		"key": "value"
		...
	}
}

GET /health

POST /login 
JSON: {
	"email": "",
	"password: ""
}
return.details.token = <JWT>

POST /register
JSON: {
	"name": "",
	"email": "",
	"password": ""
}

POST /scholarships
JSON: {
	"offset": 0,
	"limit": 5
}
return.details.rows = [<scholarships>]

POST /verify
JWt required
form-data: {
	file: <image file>,
	document_type: ""
}
return.details.model_response = <response from AI model>

POST /applications
JWT required
JSON: {
	"offset": 0,
	"limit": 5
}
return.details.rows = [<applications (either based on user, or all applications)>]

POST /dashboard
JWT required
JSON: {
	"days": 0
}
return.details.rows = [<dashboard data>]
